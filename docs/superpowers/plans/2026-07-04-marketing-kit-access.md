# Marketing Kit Access Portal Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Deliver per-app reseller landing templates on `/access/` with session-gated download and copy (HTML + plain text), sourced from 13 HTML files bundled into Cloudflare Pages Functions.

**Architecture:** HTML sources live in `marketing-kits/{appId}.html`, a Node build script emits `functions/lib/marketing-kit-content.js`, `functions/lib/marketing-kits.js` exposes lookup and plain-text extraction, `functions/api/member/marketing-kit.js` serves gated content, and `access/index.html` adds a template grid plus modal UI. Entitlement uses `computeUnlockedAppIds` identical to `/api/member/launch`.

**Tech Stack:** Cloudflare Pages Functions (ESM), Node `node:test`, Vite static build, vanilla JS in `access/index.html`.

**Spec:** `docs/superpowers/specs/2026-07-04-marketing-kit-access-design.md`

```javascript
import { readFileSync, writeFileSync, readdirSync } from 'node:fs';
import { join, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';
import { ALL_APP_IDS } from '../functions/lib/packs.js';

const root = join(dirname(fileURLToPath(import.meta.url)), '..');
const kitsDir = join(root, 'marketing-kits');
const outFile = join(root, 'functions/lib/marketing-kit-content.js');

const files = readdirSync(kitsDir).filter((f) => f.endsWith('.html'));
const map = {};
for (const file of files) {
  const appId = file.replace(/\.html$/, '');
  map[appId] = readFileSync(join(kitsDir, file), 'utf8');
}

const missing = ALL_APP_IDS.filter((id) => !map[id]);
const extra = Object.keys(map).filter((id) => !ALL_APP_IDS.includes(id));
if (missing.length) {
  console.error('Missing kits for:', missing.join(', '));
  process.exit(1);
}
if (extra.length) {
  console.warn('Extra kit files (ignored in output):', extra.join(', '));
}

const body = `// AUTO-GENERATED by scripts/bundle-marketing-kits.mjs — do not edit by hand.
export const MARKETING_KIT_HTML = ${JSON.stringify(
  Object.fromEntries(ALL_APP_IDS.map((id) => [id, map[id]])),
)};
`;
writeFileSync(outFile, body, 'utf8');
console.log('Wrote', outFile, '—', ALL_APP_IDS.length, 'kits');
```

```javascript
import { ALL_APP_IDS } from './packs.js';
import { MARKETING_KIT_HTML } from './marketing-kit-content.js';

export function hasKit(appId) {
  return ALL_APP_IDS.includes(appId) && typeof MARKETING_KIT_HTML[appId] === 'string'
    && MARKETING_KIT_HTML[appId].length > 0;
}

export function getKitHtml(appId) {
  if (!hasKit(appId)) return null;
  return MARKETING_KIT_HTML[appId];
}

export function getKitTitle(html) {
  if (!html) return '';
  const m = html.match(/<title[^>]*>([\s\S]*?)<\/title>/i);
  return m ? m[1].replace(/\s+/g, ' ').trim() : '';
}

export function extractPlainText(html) {
  if (!html) return '';
  let s = html.replace(/<script[\s\S]*?<\/script>/gi, ' ');
  s = s.replace(/<style[\s\S]*?<\/style>/gi, ' ');
  s = s.replace(/<!--[\s\S]*?-->/g, ' ');
  s = s.replace(/<\/(p|div|h[1-6]|li|br|section|header|footer|tr)>/gi, '\n');
  s = s.replace(/<br\s*\/?>/gi, '\n');
  s = s.replace(/<[^>]+>/g, ' ');
  s = s.replace(/&nbsp;/gi, ' ');
  s = s.replace(/&amp;/gi, '&');
  s = s.replace(/&lt;/gi, '<');
  s = s.replace(/&gt;/gi, '>');
  s = s.replace(/[ \t]+\n/g, '\n');
  s = s.replace(/\n{3,}/g, '\n\n');
  s = s.replace(/[ \t]{2,}/g, ' ');
  return s.trim();
}
```

```javascript
import { resolveSession } from '../../lib/session.js';
import { computeUnlockedAppIds } from '../../lib/entitlements.js';
import { ALL_APP_IDS } from '../../lib/packs.js';
import { errorResponse } from '../../lib/error-page.js';
import { getKitHtml, getKitTitle, extractPlainText } from '../../lib/marketing-kits.js';

function json(data, status = 200) {
  return new Response(JSON.stringify(data), {
    status,
    headers: { 'Content-Type': 'application/json', 'Cache-Control': 'no-store' },
  });
}

export async function onRequest(context) {
  const { request, env } = context;
  if (request.method !== 'GET') {
    return errorResponse(request, { status: 405, error: 'method_not_allowed', message: 'Metode tidak diizinkan.' });
  }

  const url = new URL(request.url);
  const appId = url.searchParams.get('app_id');
  if (!appId || !ALL_APP_IDS.includes(appId)) {
    return errorResponse(request, { status: 400, error: 'invalid_app_id', message: 'Parameter app_id tidak valid.' });
  }

  const session = await resolveSession({
    request,
    db: env.APPVIBE_DB,
    pepper: env.AUTH_TOKEN_PEPPER,
  });
  if (!session.ok) {
    return errorResponse(request, { status: session.status, error: session.error, message: session.message });
  }

  const unlocked = computeUnlockedAppIds(session.entitlements);
  if (!unlocked.includes(appId)) {
    return errorResponse(request, { status: 403, error: 'forbidden', message: 'Anda tidak memiliki akses ke aplikasi ini.' });
  }

  const html = getKitHtml(appId);
  if (!html) {
    return errorResponse(request, { status: 503, error: 'kit_not_available', message: 'Template belum tersedia.' });
  }

  const download = url.searchParams.get('download');
  if (download === '1' || download === 'true') {
    return new Response(html, {
      status: 200,
      headers: {
        'Content-Type': 'text/html; charset=utf-8',
        'Content-Disposition': `attachment; filename="${appId}-landing-template.html"`,
        'Cache-Control': 'no-store',
      },
    });
  }

  const format = url.searchParams.get('format') || 'html';
  const title = getKitTitle(html);
  if (format === 'plain') {
    return json({ app_id: appId, title, plain_text: extractPlainText(html) });
  }
  return json({ app_id: appId, title, html });
}
```

```javascript
let kitModalState = { appId: null, tab: 'html', html: '', plain: '' };

function showToast(msg) {
  const el = document.getElementById('vhToast');
  if (!el) return;
  el.textContent = msg;
  el.classList.add('show');
  setTimeout(() => el.classList.remove('show'), 2200);
}

async function openKitModal(appId) {
  const overlay = document.getElementById('vhKitModal');
  const pre = document.getElementById('vhKitModalPre');
  const title = document.getElementById('vhKitModalTitle');
  kitModalState = { appId, tab: 'html', html: '', plain: '' };
  title.textContent = `Template — ${appId}`;
  pre.textContent = 'Memuat…';
  overlay.classList.add('open');
  try {
    const [htmlRes, plainRes] = await Promise.all([
      fetch(`/api/member/marketing-kit?app_id=${encodeURIComponent(appId)}&format=html`, { credentials: 'include' }),
      fetch(`/api/member/marketing-kit?app_id=${encodeURIComponent(appId)}&format=plain`, { credentials: 'include' }),
    ]);
    const htmlData = await htmlRes.json();
    const plainData = await plainRes.json();
    kitModalState.html = htmlData.html || '';
    kitModalState.plain = plainData.plain_text || '';
    renderKitTab();
  } catch {
    pre.textContent = 'Gagal memuat template.';
  }
}

function renderKitTab() {
  const pre = document.getElementById('vhKitModalPre');
  pre.textContent = kitModalState.tab === 'plain' ? kitModalState.plain : kitModalState.html;
}

function downloadKit(appId) {
  window.location.href = `/api/member/marketing-kit?app_id=${encodeURIComponent(appId)}&download=1`;
}
```
